SECTION 1: TWO-PHASE UPLOAD FLOW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

WHY TWO PHASES:
  Vercel serverless functions have a 4.5MB request body limit. We never send
  raw file bytes through the Next.js function — only small JSON payloads.
  The actual file goes directly from the app to S3 via a presigned URL.

PHASE 1 — Get presigned URL (/api/upload-url):
  App sends { eventSlug, filename, contentType } → server returns { uploadUrl, stagingKey }
  stagingKey format: staging/{event_id}/{uuid}.{ext}

PHASE 2 — App PUTs file directly to S3:
  App uploads raw bytes to the presigned URL — Vercel is bypassed entirely.

PHASE 3 — Process (/api/upload):
  App sends { eventSlug, stagingKey, originalFilename, uploaderMobile, uploaderName, eventUserId }
  Server: fetches staged file from S3 → EXIF extraction → HEIC pre-decode →
          JPEG conversion → GPS strip → pHash → duplicate check → store
  Staging file always deleted from S3 in a finally block (success or failure).

CLIENT-SIDE TIMEOUT (lib/api.ts):
  The post() helper has a default 15s timeout. processUpload passes 60s because
  the server runs pHash + S3 move + DB write which can exceed 15s for large photos.
  All other API calls use 15s. The timeout only controls how long the app waits —
  it does NOT slow down uploads.

--- CODE: lib/api.ts --- (mobile app)
function makeTimeout(ms: number) {
  const controller = new AbortController();
  const id = setTimeout(() => controller.abort(), ms);
  return { signal: controller.signal, clear: () => clearTimeout(id) };
}

async function post(path: string, body: object, ms = 15000) {
  const { signal, clear } = makeTimeout(ms);
  try {
    const res = await fetch(`${API_BASE_URL}${path}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
      signal,
    });
    return res.json();
  } finally {
    clear();
  }
}

SECTION 2: EXIF EXTRACTION + dateSource (extractBestDate)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Every uploaded photo gets a dateSource value that controls which gallery section
it appears in and how duplicate detection treats it.

  dateSource = "exif"        → Photo Gallery (real camera photo, trusted date)
  dateSource = "file_meta"   → Other Photos Gallery (file system date, less trusted)
  dateSource = "upload_time" → Other Photos Gallery (no date info at all)

CRITICAL RULE — Make field check (added 2026-08-13):
  iOS/Android screenshots have DateTimeOriginal (time of screenshot) but NO
  camera Make field. Without this check, screenshots landed in Photo Gallery.
  Real camera photos always have Make: "Apple" / "samsung" / "OnePlus" etc.
  Now: EXIF date is only trusted if BOTH DateTimeOriginal AND Make are present.

HEIC raw fallback:
  Some iPhone HEIC files (Live Photos, burst) have 48+ iref box references.
  exifr hits a security limit and throws. We fall back to scanning the raw
  binary for TIFF magic bytes and parsing from there.

IST offset:
  EXIF DateTimeOriginal has no timezone. All events are India-based.
  We subtract 5h30m (IST = UTC+5:30) to get UTC for DB storage.

MUST be called BEFORE any format conversion touches the file.

export async function extractBestDate(filePath: string, uploadTimestamp: Date = new Date()): Promise<ExtractedDate> {
  // PRIMARY — try EXIF date taken + camera Make
  try {
    const exifData = await exifr.parse(filePath, ["DateTimeOriginal", "CreateDate", "Make"]);
    const exifDate = exifData?.DateTimeOriginal || exifData?.CreateDate;
    const hasCameraInfo = !!(exifData?.Make);
    if (exifDate instanceof Date && !isNaN(exifDate.getTime()) && hasCameraInfo) {
      const istOffsetMs = 5.5 * 60 * 60 * 1000;
      const utcDate = new Date(exifDate.getTime() - istOffsetMs);
      return { takenAt: utcDate, dateSource: "exif" };
    }
  } catch {
    // Standard parse failed — try HEIC raw TIFF scan
    try {
      const lowerPath = filePath.toLowerCase();
      if (lowerPath.endsWith(".heic") || lowerPath.endsWith(".heif")) {
        const buf = readFileSync(filePath);
        for (let i = 0; i < buf.length - 4; i++) {
          const isLE = buf[i] === 0x49 && buf[i+1] === 0x49 && buf[i+2] === 0x2A && buf[i+3] === 0x00;
          const isBE = buf[i] === 0x4D && buf[i+1] === 0x4D && buf[i+2] === 0x00 && buf[i+3] === 0x2A;
          if (isLE || isBE) {
            const tiffData = await exifr.parse(buf.slice(i), { tiff: true, exif: true, translateValues: false, reviveValues: false }).catch(() => null);
            const rawDate = tiffData?.DateTimeOriginal || tiffData?.CreateDate;
            const exifDate = rawDate instanceof Date ? rawDate : (typeof rawDate === "string" ? new Date(rawDate.replace(/^(\d{4}):(\d{2}):(\d{2})/, "$1-$2-$3")) : null);
            const hasCameraInfoHeic = !!(tiffData?.Make);
            if (exifDate && !isNaN(exifDate.getTime()) && hasCameraInfoHeic) {
              const istOffsetMs = 5.5 * 60 * 60 * 1000;
              const utcDate = new Date(exifDate.getTime() - istOffsetMs);
              return { takenAt: utcDate, dateSource: "exif" };
            }
            break;
          }
        }
      }
    } catch { /* fall through */ }
  }

  // FALLBACK 1 — file system date
  try {
    const stats = statSync(filePath);
    const fileDate = stats.birthtime && stats.birthtime.getTime() > 0 ? stats.birthtime : stats.mtime;
    if (fileDate && !isNaN(fileDate.getTime())) {
      return { takenAt: fileDate, dateSource: "file_meta" };
    }
  } catch { /* fall through */ }

  // FALLBACK 2 — upload timestamp
  return { takenAt: uploadTimestamp, dateSource: "upload_time" };
}

export function belongsInMainTimeline(dateSource: DateSource): boolean {
  return dateSource === "exif";
}

SECTION 3: HEIC PRE-DECODE + JPEG CONVERSION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

WHY:
  sharp/libvips has a 16-reference security limit for HEIC containers.
  iPhone HEIC files (Live Photos, burst shots) often have 48+ iref references.
  This causes sharp to crash. We pre-decode using heic-convert (WASM, no limits)
  BEFORE passing to sharp. EXIF is injected back into the JPEG after conversion.

  heic-convert does NOT preserve EXIF — we manually extract raw TIFF bytes from
  the HEIC box structure and inject them as an APP1 segment into the JPEG.

--- CODE: src/lib/image-processing.ts ---
export async function preDecodeIfHeic(buffer: Buffer, extension: string): Promise<Buffer> {
  if (!HEIC_EXTENSIONS.has(extension.toLowerCase())) return buffer;
  const jpegArrayBuffer = await heicConvert({ buffer: buffer as unknown as ArrayBuffer, format: "JPEG", quality: 1 });
  const jpegBuffer = Buffer.from(jpegArrayBuffer);
  return injectExifFromHeic(buffer, jpegBuffer);
}

export async function convertToJpg(inputBuffer: Buffer): Promise<Buffer> {
  return sharp(inputBuffer)
    .rotate()
    .jpeg({ quality: 95, mozjpeg: true })
    .withMetadata({ orientation: 1 })
    .toBuffer();
}

--- CODE: src/lib/image-processing.ts ---
export function stripGpsExif(jpegBuffer: Buffer): Buffer {
  try {
    const dataUrl = "data:image/jpeg;base64," + jpegBuffer.toString("base64");
    let exifObj: any;
    try { exifObj = piexif.load(dataUrl); } catch { return jpegBuffer; }
    exifObj["GPS"] = {};
    const exifBytes = piexif.dump(exifObj);
    const result = piexif.insert(exifBytes, dataUrl);
    return Buffer.from(result.replace("data:image/jpeg;base64,", ""), "base64");
  } catch {
    return jpegBuffer;
  }
}

SECTION 5: DUPLICATE DETECTION (isDuplicate)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Uses perceptual hashing (pHash via sharp-phash). Detects visually identical photos
even when one is recompressed (e.g. WhatsApp copies). Logic lives in:
  src/lib/duplicate-detection.ts

THRESHOLDS:
  HASH_THRESHOLD = 10          — outer limit, beyond this never a duplicate
  HASH_THRESHOLD_STRICT = 4    — tight match, almost certainly same photo
  SIZE_SIMILARITY_RATIO = 15%  — files within 15% size = "same quality"
  MIN_WHATSAPP_DELAY_MS = 10000 — minimum gap to treat as WhatsApp timestamp shift
    (burst shots always under 10s apart; WhatsApp delivery takes longer)

isDuplicate() RULES (checked in order):
  1. If hash distance > 10 → NOT duplicate
  2. If distance 5–10 (loose band):
     Match ONLY if exactly one side has EXIF (WhatsApp copy scenario)
     OR both have EXIF + timestamps within 2s + sizes differ >15% (WhatsApp HD)
     Otherwise → NOT duplicate
  3. If both have EXIF and timestamps differ by >2s → NOT duplicate
     Exception: if distance ≤4 AND sizes differ >15% AND gap >10s → WhatsApp
     shifted timestamp to receive time; trust the visual hash
  4. Hash collision guard: if one has EXIF, size diff >90%, distance >4 → NOT duplicate
  5. Neither has EXIF: only match if distance === 0 (pixel-perfect identical)
  6. Otherwise → IS duplicate

--- CODE: src/lib/duplicate-detection.ts ---
const HASH_THRESHOLD = 10;
const HASH_THRESHOLD_STRICT = 4;
const SIZE_SIMILARITY_RATIO = 0.15;
const MIN_WHATSAPP_DELAY_MS = 10000;

export function isDuplicate(
  hashA: string, hashB: string,
  sizeA: number, sizeB: number,
  takenAtA?: Date | null, takenAtB?: Date | null,
  dateSourceA?: string | null, dateSourceB?: string | null,
): boolean {
  const distance = hammingDistance(hashA, hashB);
  if (distance > HASH_THRESHOLD) return false;

  const oneHasExif = (dateSourceA === "exif") !== (dateSourceB === "exif");
  const bothHaveExif = dateSourceA === "exif" && dateSourceB === "exif";
  const timestampsClose = takenAtA != null && takenAtB != null &&
    Math.abs(takenAtA.getTime() - takenAtB.getTime()) <= 2000;
  const sizeDiffRatio = Math.abs(sizeA - sizeB) / Math.max(sizeA, sizeB);
  const sizeDiffSignificant = sizeDiffRatio > SIZE_SIMILARITY_RATIO;

  if (distance > HASH_THRESHOLD_STRICT) {
    const isWhatsappHd = bothHaveExif && timestampsClose && sizeDiffSignificant;
    if (!oneHasExif && !isWhatsappHd) return false;
  }

  const tsGapMs = takenAtA != null && takenAtB != null
    ? Math.abs(takenAtA.getTime() - takenAtB.getTime()) : null;
  const whatsappTimestampShift = distance <= HASH_THRESHOLD_STRICT && sizeDiffSignificant
    && tsGapMs != null && tsGapMs > MIN_WHATSAPP_DELAY_MS;

  if (bothHaveExif && !timestampsClose && takenAtA != null && takenAtB != null && !whatsappTimestampShift) {
    return false;
  }

  if (oneHasExif && sizeDiffRatio > 0.90 && distance > HASH_THRESHOLD_STRICT) return false;
  if (oneHasExif) return true;
  if (bothHaveExif && timestampsClose && sizeDiffSignificant) return true;
  if (whatsappTimestampShift) return true;
  if (bothHaveExif && timestampsClose && !sizeDiffSignificant && distance > 0) return false;

  const neitherHasExif = dateSourceA !== "exif" && dateSourceB !== "exif";
  if (neitherHasExif) return distance === 0;

  return sizeDiffRatio <= SIZE_SIMILARITY_RATIO;
}

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

WHEN UPGRADE HAPPENS:
  — New file is larger (less compressed) → new wins
  — New file has genuine EXIF date, existing only has fallback → new wins
  — Common: WhatsApp copy uploaded first (small, no EXIF), then original (large, EXIF)

WHAT HAPPENS ON UPGRADE (server-side):
  1. Convert new file, strip GPS, generate new thumbnail + display
  2. Upload all 3 new versions to S3
  3. UPDATE existing photo DB row IN-PLACE (same photo ID, same timeline position)
  4. Notify original uploader if they are a different person

WHY IN-PLACE UPDATE:
  Preserves photo's position in event timeline and keeps the same photo ID.
  The WhatsApp copy "becomes" the original.

CRITICAL: Do NOT modify this logic without explicit user approval.

--- CODE: src/lib/duplicate-detection.ts ---
export function resolveDuplicate(existing: PhotoCandidate, incoming: NewUpload): DuplicateResolution {
  const sizeDiff = incoming.fileSizeBytes - existing.fileSizeBytes;
  const sameQuality = Math.abs(sizeDiff) < existing.fileSizeBytes * 0.02;
  const sameDateSource = incoming.dateSource === existing.dateSource;
  const sameTimestamp = Math.abs(incoming.takenAt.getTime() - existing.takenAt.getTime()) < 1000;

  if (sameQuality && sameDateSource && sameTimestamp) {
    return { action: "merge_self_duplicate", reason: "Identical quality and timestamp." };
  }

  const incomingHigherQuality = sizeDiff > 0 && !sameQuality;
  const incomingMoreTrustworthyDate = incoming.dateSource === "exif" && existing.dateSource !== "exif";

  if (incomingHigherQuality || incomingMoreTrustworthyDate) {
    return { action: "keep_new", reason: incomingHigherQuality ? "New upload is higher quality." : "New upload has genuine EXIF date." };
  }

  return { action: "keep_existing", reason: "Existing photo is equal or higher quality." };
}

Android uses modules/media-store — wraps Android's MediaStore API.

SINGLE PHOTO DOWNLOAD:
  1. Download photo from server to Expo cache directory
  2. Pass local path to MediaStore.saveToDownloads(path, filename, folderName, mimeType, dateTakenMs)
  3. MediaStore saves to Downloads/[EventFolderName]/filename.jpg
  4. dateTakenMs sets correct date in Android's media database
  5. Delete temp cache file after saving

ZIP DOWNLOAD (bulk, >25 photos):
  Same flow but mimeType = 'application/zip'. Split into batches if needed.

KEY DIFFERENCE FROM iOS:
  — Android reads date from dateTakenMs passed to MediaStore (no EXIF embed needed)
  — iOS ignores creation date passed in code — must embed EXIF into file
  — Android saves to Downloads folder (file manager accessible)
  — iOS saves to Photos app (camera roll / named album)
  — iOS ZIP: shared via iOS Share Sheet (no Downloads folder for apps)

FOLDER NAMING:
  First download: user asked to name folder (defaults to event name).
  Stored in SecureStore keyed by event slug. Never asked again for same event.

const MediaStore = Platform.OS === 'android' ? require('media-store').default : null;

async function saveFileToDownloads(filename, url, mimeType, folderPath, mode, notify = false, dateTakenMs?) {
  if (Platform.OS === 'android') {
    const folderName = await SecureStore.getItemAsync(`downloads_folder_name_${slug}`) ?? params.name;
    const cacheUri = `${FileSystem.cacheDirectory}${filename}`;
    const dlResult = await FileSystem.downloadAsync(url, cacheUri);
    if (dlResult.status !== 200) throw new Error(`HTTP ${dlResult.status}`);
    const localPath = dlResult.uri.replace('file://', '');
    await MediaStore.saveToDownloads(localPath, filename, folderName, mimeType, dateTakenMs);
    await FileSystem.deleteAsync(cacheUri, { idempotent: true });
  } else {
    const cacheUri = `${FileSystem.cacheDirectory}${filename}`;
    const dlResult = await FileSystem.downloadAsync(url, cacheUri);
    if (dlResult.status !== 200) throw new Error(`HTTP ${dlResult.status}`);
    if (PhotoSaver) {
      await PhotoSaver.saveToPhotos(cacheUri, dateTakenMs ?? 0, params.name);
    } else {
      await MediaLibrary.saveToLibraryAsync(cacheUri);
    }
    await FileSystem.deleteAsync(cacheUri, { idempotent: true });
  }
}

export async function DELETE(request, { params }) {
  const { phone } = await params;
  const { callerPhone, password } = await request.json();

  if (phone === MASTER_PHONE) {
    return NextResponse.json({ error: "Cannot remove the master number." }, { status: 400 });
  }

  if (!await verifyMaster(callerPhone, password)) {
    return NextResponse.json({ error: "Unauthorised." }, { status: 403 });
  }

  // Find all events owned by this phone
  const { data: events } = await supabaseAdmin.from("events").select("id").eq("owner_phone", phone);
  const eventIds = (events ?? []).map(e => e.id);

  if (eventIds.length > 0) {
    // Fetch all photo S3 paths
    const { data: photos } = await supabaseAdmin.from("photos")
      .select("storage_path, thumbnail_path, display_path").in("event_id", eventIds);

    if (photos?.length) {
      const keys = photos.flatMap(p => [p.storage_path, p.thumbnail_path, p.display_path]).filter(Boolean);
      // Delete S3 objects in batches of 1000
      for (let i = 0; i < keys.length; i += 1000) {
        await s3Client.send(new DeleteObjectsCommand({
          Bucket: S3_BUCKET,
          Delete: { Objects: keys.slice(i, i + 1000).map(Key => ({ Key })) },
        })).catch(() => {});
      }
    }

    // Delete photo DB rows, then events (CASCADE handles related tables)
    await supabaseAdmin.from("photos").delete().in("event_id", eventIds);
    await supabaseAdmin.from("events").delete().eq("owner_phone", phone);
  }

  // Delete from whitelist
  await supabaseAdmin.from("organiser_whitelist").delete().eq("phone", phone);
  return NextResponse.json({ success: true });
}

Every local native module in modules/ MUST have an entry in the root package.json
or requireNativeModule returns null at runtime.

  "dependencies": {
    "photo-saver": "file:./modules/photo-saver",
    "media-store": "file:./modules/media-store"

Missing this entry → JS loads fine → native module is null → crash at call site.
